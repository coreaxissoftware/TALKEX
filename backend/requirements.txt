# Versions are floors with a capped ceiling, not exact pins. The original
# file pinned pydantic==2.10.6, which has no wheel for Python 3.14 and tries
# to compile pydantic-core from Rust source on install — that fails on a
# machine without a Rust toolchain. An open floor-only range has the
# opposite problem: every `pip install -r requirements.txt` (every Render
# deploy, per render.yaml's buildCommand) re-resolves whatever the newest
# matching release is AT BUILD TIME, so a breaking major/minor release can
# reach production on the next deploy without ever having run against
# test_api.py first. The ceilings below allow patch/compatible upgrades —
# the actual point of not exact-pinning — while blocking an unvetted jump
# across a version boundary that's historically where breaking changes live.
fastapi>=0.115,<1.0
uvicorn[standard]>=0.30,<1.0
websockets>=12.0,<17.0
python-multipart>=0.0.9,<1.0
pydantic>=2.11,<3.0

# Used by the test suite (fastapi.testclient needs httpx).
httpx>=0.27,<1.0
pytest>=8.0,<10.0

# Renders the device-linking QR code server-side (SVG, no Pillow needed) —
# a trusted, spec-correct implementation rather than a hand-rolled encoder.
qrcode>=7.4,<9.0

# Sends Web Push notifications (VAPID + the browser Push API), so a message
# can reach someone whose tab isn't open, not just whoever has a live socket.
pywebpush>=2.0,<3.0

# Mints the short-lived OAuth token the FCM HTTP v1 API needs from the
# service-account key, so the backend can push to native (Android) devices —
# the Capacitor app's WebView can't receive Web Push, only FCM.
google-auth>=2.0,<3.0
